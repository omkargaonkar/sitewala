  if (!errorMsg) return;

  // ensure ID exists
  if (!errorMsg.id) {
    errorMsg.id = 'edit-field-ar-error';
  }

  const inputs = wrapper.querySelectorAll('input[type="checkbox"]');

  inputs.forEach(input => {
    input.setAttribute('aria-invalid', 'true');
    input.setAttribute('aria-describedby', errorMsg.id);
  });
})();
